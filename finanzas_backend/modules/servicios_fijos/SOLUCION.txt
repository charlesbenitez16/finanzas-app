GUÍA DE SOLUCIÓN — MÓDULO servicios_fijos
================================================================================
Entidad SERVICIOS_FIJOS / tabla `servicios_fijos`. Es la "plantilla" de un
gasto recurrente (Internet, Agua, Luz, Gimnasio). Cada mes genera un pago
real que se registra como Gasto en el módulo gastos. Módulo propio porque
tiene ciclo de vida independiente (activo, fecha_inicio/fin, config de
alerta).

1. domain/entities.py  ->  ServicioFijo   (el corazón del módulo)

    from calendar import monthrange
    from datetime import date

    def esta_activo(self) -> bool:
        return self.activo

    def esta_vigente(self, fecha: date) -> bool:
        if not self.activo:
            return False
        if self.fecha_inicio and fecha < self.fecha_inicio:
            return False
        if self.fecha_fin and fecha > self.fecha_fin:
            return False
        return True

    def fecha_vencimiento_en(self, mes: int, anio: int) -> date:
        ultimo_dia = monthrange(anio, mes)[1]
        return date(anio, mes, min(self.dia_vencimiento, ultimo_dia))

    def dias_para_vencimiento(self, fecha_ref: date) -> int:
        venc = self.fecha_vencimiento_en(fecha_ref.month, fecha_ref.year)
        return (venc - fecha_ref).days

    def debe_generar_alerta(self, fecha_ref: date) -> bool:
        if not self.esta_vigente(fecha_ref):
            return False
        return self.dias_para_vencimiento(fecha_ref) == self.dias_anticipacion_alerta

POR QUÉ:
- esta_activo() / esta_vigente(): la primera es el flag pelado; la segunda
  combina el flag con la ventana [fecha_inicio, fecha_fin] (fecha_fin None
  = sin límite). Todo son atributos propios + un parámetro fecha -> es
  responsabilidad de la entidad.
- fecha_vencimiento_en: DECISIÓN sobre dia_vencimiento = 31 en meses de
  28/29/30 días. monthrange(anio, mes)[1] da la cantidad de días del mes;
  min(dia_vencimiento, ultimo_dia) -> política "si el día no existe, vence
  el último día del mes". Alternativa: pasar al 1° del mes siguiente
  (peor: adelanta el vencimiento). Dejá documentada la que elijas.
- dias_para_vencimiento: mira el vencimiento DE ESTE MES respecto de
  fecha_ref. Puede dar negativo si fecha_ref ya pasó el día. Es
  equivalente a `(dia_vencimiento - EXTRACT(DAY FROM CURRENT_DATE))` de la
  query comentada en schema_finanzas.sql.
- debe_generar_alerta: es la parte "de la entidad" del generador de
  alertas (vista E). "Faltan EXACTAMENTE dias_anticipacion_alerta días" +
  "está vigente". El "y además no se pagó este mes" lo agrega el caso de
  uso del módulo alertas (necesita el GastoRepository).
- OJO límite de mes: la fórmula del SQL (y esta) falla si el día de
  vencimiento es chico y fecha_ref está a fin de mes (ej. vence el 2, hoy
  es 30). Para el kata replicamos el SQL; si querés arreglarlo, en
  dias_para_vencimiento compará también contra el vencimiento del mes
  siguiente y tomá el menor positivo.

    class DjangoServicioFijoRepository(ServicioFijoRepository):
        def _a_entidad(self, m: ServicioFijoModel) -> ServicioFijo:
            return ServicioFijo(
                id=m.id, usuario_id=m.usuario_id, categoria_id=m.categoria_id,
                prioridad_id=m.prioridad_id, nombre=m.nombre,
                dia_vencimiento=m.dia_vencimiento, monto_estimado=m.monto_estimado,
                es_monto_variable=m.es_monto_variable,
                dias_anticipacion_alerta=m.dias_anticipacion_alerta, activo=m.activo,
                fecha_inicio=m.fecha_inicio, fecha_fin=m.fecha_fin, notas=m.notas,
            )

        def obtener_por_id(self, servicio_id):
            try:
                return self._a_entidad(ServicioFijoModel.objects.get(id=servicio_id))
            except ServicioFijoModel.DoesNotExist:
                return None

        def crear(self, servicio):
            m = ServicioFijoModel.objects.create(
                usuario_id=servicio.usuario_id, categoria_id=servicio.categoria_id,
                prioridad_id=servicio.prioridad_id, nombre=servicio.nombre,
                dia_vencimiento=servicio.dia_vencimiento,
                monto_estimado=servicio.monto_estimado,
                es_monto_variable=servicio.es_monto_variable,
                dias_anticipacion_alerta=servicio.dias_anticipacion_alerta,
                activo=servicio.activo, fecha_inicio=servicio.fecha_inicio,
                fecha_fin=servicio.fecha_fin, notas=servicio.notas,
            )
            return self._a_entidad(m)

        def actualizar(self, servicio):
            ServicioFijoModel.objects.filter(id=servicio.id).update(
                nombre=servicio.nombre, dia_vencimiento=servicio.dia_vencimiento,
                monto_estimado=servicio.monto_estimado,
                es_monto_variable=servicio.es_monto_variable,
                dias_anticipacion_alerta=servicio.dias_anticipacion_alerta,
                activo=servicio.activo, fecha_inicio=servicio.fecha_inicio,
                fecha_fin=servicio.fecha_fin, notas=servicio.notas,
            )
            return servicio

        def listar_por_usuario(self, usuario_id, solo_activos=False):
            qs = ServicioFijoModel.objects.filter(usuario_id=usuario_id)
            if solo_activos:
                qs = qs.filter(activo=True)
            return [self._a_entidad(m) for m in qs.order_by("nombre")]

        def listar_proximos_a_vencer(self, fecha_referencia):
            # OPCIÓN B (recomendada): traer activos y filtrar con la
            # lógica de dominio -> la regla vive en UN solo lugar (la entidad).
            activos = [self._a_entidad(m)
                       for m in ServicioFijoModel.objects.filter(activo=True)]
            return [s for s in activos if s.debe_generar_alerta(fecha_referencia)]

    from datetime import date

    def ejecutar(self, usuario_id, categoria_id, prioridad_id, nombre,
                 dia_vencimiento, monto_estimado=None, es_monto_variable=False,
                 dias_anticipacion_alerta=3, fecha_inicio=None, fecha_fin=None,
                 notas=None):
        if not (1 <= dia_vencimiento <= 31):
            raise DiaVencimientoInvalido("dia_vencimiento debe estar entre 1 y 31")
        if monto_estimado is not None and monto_estimado < 0:
            raise MontoEstimadoInvalido("monto_estimado no puede ser negativo")
        fecha_inicio = fecha_inicio or date.today()
        if fecha_fin is not None and fecha_fin < fecha_inicio:
            raise RangoFechasInvalido("fecha_fin es anterior a fecha_inicio")
        servicio = ServicioFijo(
            usuario_id=usuario_id, categoria_id=categoria_id,
            prioridad_id=prioridad_id, nombre=nombre.strip(),
            dia_vencimiento=dia_vencimiento, monto_estimado=monto_estimado,
            es_monto_variable=es_monto_variable,
            dias_anticipacion_alerta=dias_anticipacion_alerta, activo=True,
            fecha_inicio=fecha_inicio, fecha_fin=fecha_fin, notas=notas,
        )
        return self.repositorio.crear(servicio)

    class ServiciosFijosView(APIView):
        def get(self, request):
            caso = ListarServiciosActivosUseCase(repositorio=DjangoServicioFijoRepository())
            return Response(ServicioFijoSerializer(
                caso.ejecutar(request.user.id), many=True).data)

        def post(self, request):
            entrada = ServicioFijoSerializer(data=request.data)
            entrada.is_valid(raise_exception=True)
            caso = CrearServicioFijoUseCase(repositorio=DjangoServicioFijoRepository())
            try:
                servicio = caso.ejecutar(
                    usuario_id=request.user.id,
                    categoria_id=entrada.validated_data["categoria_id"],
                    prioridad_id=entrada.validated_data["prioridad_id"],
                    nombre=entrada.validated_data["nombre"],
                    dia_vencimiento=entrada.validated_data["dia_vencimiento"],
                    monto_estimado=entrada.validated_data.get("monto_estimado"),
                    es_monto_variable=entrada.validated_data.get("es_monto_variable", False),
                    dias_anticipacion_alerta=entrada.validated_data.get("dias_anticipacion_alerta", 3),
                    fecha_inicio=entrada.validated_data.get("fecha_inicio"),
                    fecha_fin=entrada.validated_data.get("fecha_fin"),
                    notas=entrada.validated_data.get("notas"),
                )
            except (DiaVencimientoInvalido, MontoEstimadoInvalido, RangoFechasInvalido) as e:
                return Response({"detail": str(e)}, status=status.HTTP_400_BAD_REQUEST)
            return Response(ServicioFijoSerializer(servicio).data, status=status.HTTP_201_CREATED)

    def test_debe_generar_alerta_a_los_dias_de_anticipacion_exactos(self):
        # vence el 10, avisa 3 días antes -> parado el 7 debe alertar
        s = ServicioFijo(dia_vencimiento=10, dias_anticipacion_alerta=3, activo=True,
                         fecha_inicio=date(2026, 1, 1))
        self.assertTrue(s.debe_generar_alerta(date(2026, 3, 7)))
        self.assertFalse(s.debe_generar_alerta(date(2026, 3, 6)))

    def test_esta_vigente_false_fuera_de_la_ventana(self):
