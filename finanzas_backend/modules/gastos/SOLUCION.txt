    class DjangoGastoRepository(GastoRepository):
        def _a_entidad(self, m: GastoModel) -> Gasto:
            return Gasto(
                id=m.id, usuario_id=m.usuario_id, categoria_id=m.categoria_id,
                servicio_fijo_id=m.servicio_fijo_id, prioridad_id=m.prioridad_id,
                monto=m.monto, fecha_gasto=m.fecha_gasto, periodo_mes=m.periodo_mes,
                periodo_anio=m.periodo_anio, estado=m.estado,
                descripcion=m.descripcion, fecha_registro=m.fecha_registro,
            )

        def obtener_por_id(self, gasto_id):
            try:
                return self._a_entidad(GastoModel.objects.get(id=gasto_id))
            except GastoModel.DoesNotExist:
                return None

        def crear(self, gasto):
            m = GastoModel.objects.create(
                usuario_id=gasto.usuario_id, categoria_id=gasto.categoria_id,
                servicio_fijo_id=gasto.servicio_fijo_id,
                prioridad_id=gasto.prioridad_id, monto=gasto.monto,
                fecha_gasto=gasto.fecha_gasto, periodo_mes=gasto.periodo_mes,
                periodo_anio=gasto.periodo_anio, estado=gasto.estado,
                descripcion=gasto.descripcion,
            )
            return self._a_entidad(m)

        def listar_por_usuario_y_periodo(self, usuario_id, periodo_mes, periodo_anio):
            qs = GastoModel.objects.filter(
                usuario_id=usuario_id, periodo_mes=periodo_mes,
                periodo_anio=periodo_anio,
            ).order_by("fecha_gasto")
            return [self._a_entidad(m) for m in qs]

        def sumar_montos_por_periodo(self, usuario_id, periodo_mes, periodo_anio):
            total = GastoModel.objects.filter(
                usuario_id=usuario_id, periodo_mes=periodo_mes,
                periodo_anio=periodo_anio,
            ).aggregate(t=Sum("monto"))["t"]
            return total or Decimal("0")

        def existe_pago_de_servicio_en_periodo(self, servicio_fijo_id, periodo_mes, periodo_anio):
            return GastoModel.objects.filter(
                servicio_fijo_id=servicio_fijo_id, periodo_mes=periodo_mes,
                periodo_anio=periodo_anio,
            ).exists()

        def listar_pagos_de_servicio_fijo(self, servicio_fijo_id):
            qs = GastoModel.objects.filter(
                servicio_fijo_id=servicio_fijo_id,
            ).order_by("periodo_anio", "periodo_mes")
            return [self._a_entidad(m) for m in qs]

        def total_por_prioridad(self, usuario_id, periodo_mes, periodo_anio):
            filas = (
                GastoModel.objects
                .filter(usuario_id=usuario_id, periodo_mes=periodo_mes,
                        periodo_anio=periodo_anio)
                .values("prioridad__nombre", "prioridad__nivel_orden")
                .annotate(total=Sum("monto"))
                .order_by("prioridad__nivel_orden")
            )
            return [
                AhorroPorPrioridad(
                    prioridad=f["prioridad__nombre"],
                    nivel_orden=f["prioridad__nivel_orden"],
                    total_por_prioridad=f["total"],
                )
                for f in filas
            ]

POR QUÉ:
- .values(...).annotate(Sum(...)) es exactamente un GROUP BY. El
  order_by("prioridad__nivel_orden") reproduce el ORDER BY de
  vista_ahorro_potencial_por_prioridad.
- prioridad__nombre atraviesa la FK a catalogos.NivelPrioridadModel: hace
  el JOIN que en el SQL es `JOIN niveles_prioridad np ON np.id = g.prioridad_id`.
  (Si hubieras usado IntegerField en vez de FK: .values("prioridad_id")

- listar_pagos_de_servicio_fijo ORDENA por (anio, mes): es el insumo del
  LAG del punto 6. El orden lo hace el repo, no el caso de uso.

7. use_case: calcular_variacion_servicios  (vista B — el LAG)
--------------------------------------------------------------------------------
    from decimal import Decimal

    def ejecutar(self, servicio_fijo_id):
        pagos = self.repositorio.listar_pagos_de_servicio_fijo(servicio_fijo_id)
        resultado = []
        anterior = None
        for pago in pagos:                      # ya vienen ordenados por (anio, mes)
            if anterior is None or anterior.monto == 0:
                pct = None
            else:
                pct = ((pago.monto - anterior.monto) / anterior.monto
                       * Decimal("100")).quantize(Decimal("0.01"))
            resultado.append(VariacionServicio(
                servicio_fijo_id=servicio_fijo_id,
                servicio="",                    # ver nota
                periodo_mes=pago.periodo_mes, periodo_anio=pago.periodo_anio,
                monto_actual=pago.monto,
                monto_anterior=(anterior.monto if anterior else None),
                porcentaje_variacion=pct,
            ))
            anterior = pago
        return resultado

POR QUÉ:
- Esto ES el `LAG(monto) OVER (PARTITION BY servicio_fijo_id ORDER BY
  periodo_anio, periodo_mes)` de vista_variacion_servicios, hecho en
  Python. `anterior` = la "fila previa" que el LAG te da.
- Primer pago -> no hay anterior -> monto_anterior y pct en None. División
  por cero (anterior.monto == 0) -> pct None. Eso reproduce el
  `NULLIF(LAG(...), 0)` del SQL.
- .quantize(Decimal("0.01")) = ROUND(..., 2).
- Ejemplo del schema: Agua 5000 -> 5500 => (5500-5000)/5000*100 = +10.00.
- CAMPO `servicio` (nombre): el Gasto no lo trae. Opción recomendada:
  inyectá ServicioFijoRepository, hacé una sola llamada
  obtener_por_id(servicio_fijo_id) y usá .nombre para todas las filas.
- ALTERNATIVA pura ORM (función ventana de Django):
      from django.db.models import Window, F
      from django.db.models.functions import Lag
      qs = (GastoModel.objects.filter(servicio_fijo_id=servicio_fijo_id)
            .annotate(monto_anterior=Window(
                expression=Lag("monto"),
                order_by=[F("periodo_anio").asc(), F("periodo_mes").asc()]))
            .order_by("periodo_anio", "periodo_mes"))
  y calculás el pct en Python. El loop es más claro para el kata.
- OPCIONAL: implementar shared/domain/value_objects.py::Dinero
  (porcentaje_de_variacion_respecto_a) y delegar ahí el cálculo.

8. use_case: simular_modo_emergencia  (vista C + acumulado)
--------------------------------------------------------------------------------
    from decimal import Decimal
    from modules.gastos.domain.entities import AhorroEmergencia

    def ejecutar(self, usuario_id, periodo_mes, periodo_anio):
        gastos = {
            fila.nivel_orden: fila.total_por_prioridad
            for fila in self.gasto_repositorio.total_por_prioridad(
                usuario_id, periodo_mes, periodo_anio)
        }
        niveles = {n.nivel_orden: n
                   for n in self.nivel_prioridad_repositorio.listar_todos()}

        acumulado = Decimal("0")
        filas = []
        # de MÁS prescindible (mayor nivel_orden) a MÁS esencial (1)
        for orden in sorted(niveles, reverse=True):
            total = gastos.get(orden, Decimal("0"))
            acumulado += total
            filas.append(AhorroEmergencia(
                prioridad=niveles[orden].nombre,
                nivel_orden=orden,
                total_periodo=total,
                ahorro_acumulado_si_suspende=acumulado,
            ))
        filas.reverse()                       # volver al orden natural (1..3)
        return filas

POR QUÉ:
- Es EXACTAMENTE el caso de uso que catalogos anticipó en el comentario de
  NivelPrioridad: "un NivelPrioridad aislado no sabe si es el más bajo;
  hay que ver TODOS los niveles a la vez". Por eso se inyecta
  nivel_prioridad_repositorio y se hace listar_todos().
- vista_ahorro_potencial_por_prioridad da SUM(monto) por prioridad. Acá
  agregamos el valor del "modo emergencia": el ahorro ACUMULADO si vas
  suspendiendo desde lo más prescindible hacia arriba
  ("suspendo Prescindible: ahorro X; + Importante: X+Y; ...").
- Usar niveles.listar_todos() (y no solo las prioridades con gasto) hace
  que una prioridad sin gastos este mes igual aparezca con total 0 -> el
  reporte queda completo.
- DECISIÓN: ¿incluís nivel_orden == 1 (Esencial) en el acumulado? En
  emergencia normalmente NO se suspende lo esencial. Si querés excluirlo:
  `for orden in sorted(niveles, reverse=True): if orden == 1: break`
  o dejá la fila pero sin sumar. Documentá la elección.
