GUÍA DE SOLUCIÓN — MÓDULO alertas
================================================================================
Entidad ALERTAS_PAGO / tabla `alertas_pago`. Recordatorios de vencimiento
de servicios fijos, con máquina de estados
  Pendiente -> Enviada -> Leida -> Resuelta
Dos piezas especiales:
  - GenerarAlertasDeVencimientoUseCase: el job/cron (traduce la query
    comentada al final de schema_finanzas.sql).
  - ResolverAlertaAlPagarUseCase: HANDLER del evento GastoRegistrado que
    publica el módulo gastos (relación `GASTOS |o--o| ALERTAS_PAGO` del .mermaid).

Requisito de infra: shared/domain/events.py::EventBus implementado
(ver CONFIGURACION_FUNCIONAL.txt sección 6.2).


--------------------------------------------------------------------------------
1. domain/entities.py  ->  AlertaPago  (máquina de estados)
--------------------------------------------------------------------------------
    _TRANSICIONES = {
        "Pendiente": {"Enviada"},
        "Enviada": {"Leida"},
        "Leida": set(),
        "Resuelta": set(),
    }

    def esta_pendiente(self) -> bool:
        return self.estado == "Pendiente"

    def esta_resuelta(self) -> bool:
        return self.estado == "Resuelta"

    def _transicionar(self, nuevo: str) -> None:
        if nuevo not in _TRANSICIONES.get(self.estado, set()):
            raise TransicionEstadoInvalida(
                f"No se puede pasar de {self.estado} a {nuevo}")
        self.estado = nuevo

    def marcar_enviada(self) -> None:
        self._transicionar("Enviada")

    def marcar_leida(self) -> None:
        self._transicionar("Leida")

    def resolver(self, gasto_id: int) -> None:
        if self.esta_resuelta():
            return                       # idempotente: el evento puede llegar 2 veces
        self.gasto_id = gasto_id
        self.estado = "Resuelta"

    def es_estado_valido(self) -> bool:
        return self.estado in ESTADOS_ALERTA

POR QUÉ:
- Tabla de transiciones = el grafo legal en UN solo lugar. marcar_enviada
  / marcar_leida solo la consultan. Transición ilegal -> excepción de
  dominio.
- resolver() es especial: puede dispararse desde CUALQUIER estado no
  resuelto (pagar siempre resuelve, no importa si la alerta estaba
  Pendiente o Leida) y es idempotente (si el bus entrega el evento dos
  veces, la segunda no hace nada). Por eso NO pasa por _transicionar().
- es_estado_valido() replica el CHECK
  (estado IN ('Pendiente','Enviada','Leida','Resuelta')).
- La entidad SÍ es dueña de su máquina de estados: solo mira self.estado,
  no necesita nada externo (a diferencia de NivelPrioridad en catalogos).

    from typing import Optional

    def _a_entidad(self, m):
        return AlertaPago(
            id=m.id, usuario_id=m.usuario_id, servicio_fijo_id=m.servicio_fijo_id,
            gasto_id=m.gasto_id, periodo_mes=m.periodo_mes, periodo_anio=m.periodo_anio,
            fecha_alerta=m.fecha_alerta, estado=m.estado, mensaje=m.mensaje,
            fecha_creacion=m.fecha_creacion,
        )

    def obtener_por_id(self, alerta_id):
        try:
            return self._a_entidad(AlertaPagoModel.objects.get(id=alerta_id))
        except AlertaPagoModel.DoesNotExist:
            return None

    def crear(self, alerta):
        m = AlertaPagoModel.objects.create(
            usuario_id=alerta.usuario_id, servicio_fijo_id=alerta.servicio_fijo_id,
            gasto_id=alerta.gasto_id, periodo_mes=alerta.periodo_mes,
            periodo_anio=alerta.periodo_anio, fecha_alerta=alerta.fecha_alerta,
            estado=alerta.estado, mensaje=alerta.mensaje,
        )
        return self._a_entidad(m)

    def actualizar(self, alerta):
        AlertaPagoModel.objects.filter(id=alerta.id).update(
            gasto_id=alerta.gasto_id, estado=alerta.estado, mensaje=alerta.mensaje,
        )
        return alerta

    def listar_por_usuario_y_estado(self, usuario_id, estado):
        qs = AlertaPagoModel.objects.filter(
            usuario_id=usuario_id, estado=estado,
        ).order_by("fecha_alerta")
        return [self._a_entidad(m) for m in qs]

    def existe_alerta_para_servicio_en_periodo(self, servicio_fijo_id, periodo_mes, periodo_anio):
        return AlertaPagoModel.objects.filter(
            servicio_fijo_id=servicio_fijo_id, periodo_mes=periodo_mes,
            periodo_anio=periodo_anio,
        ).exists()

    def obtener_por_servicio_y_periodo(self, servicio_fijo_id, periodo_mes, periodo_anio):
        m = AlertaPagoModel.objects.filter(
            servicio_fijo_id=servicio_fijo_id, periodo_mes=periodo_mes,
            periodo_anio=periodo_anio,
        ).first()
        return self._a_entidad(m) if m else None

    def ejecutar(self, fecha_referencia=None):
        fecha_referencia = fecha_referencia or date.today()
        servicios = self.servicio_fijo_repositorio.listar_proximos_a_vencer(fecha_referencia)

4.4 resolver_alerta_al_pagar.py   (HANDLER del evento)

    import logging
    logger = logging.getLogger(__name__)

    def manejar(self, evento):
        if evento.servicio_fijo_id is None:
            return                                   # gasto puntual: no hay alerta
        try:
            alerta = self.repositorio.obtener_por_servicio_y_periodo(
                evento.servicio_fijo_id, evento.periodo_mes, evento.periodo_anio)
            if alerta is None or alerta.esta_resuelta():
                return
            alerta.resolver(evento.gasto_id)
            self.repositorio.actualizar(alerta)
        except Exception:                            # noqa: BLE001
            logger.exception("No se pudo resolver la alerta para el evento %r", evento)

  POR QUÉ tragar la excepción: este handler corre SÍNCRONO dentro de
  RegistrarPagoDeServicioFijoUseCase (via event_bus.publicar). Si el
  update de la alerta falla, NO debe voltear ni romper el registro del
  pago. Se loguea y sigue. (En un bus async con reintentos, dejarías

    from datetime import date
    from modules.servicios_fijos.infrastructure.repositories import DjangoServicioFijoRepository

    class AlertasView(APIView):
        def get(self, request):
            estado = request.query_params.get("estado", "Pendiente")
            caso = ListarAlertasPendientesUseCase(repositorio=DjangoAlertaPagoRepository())
            return Response(AlertaPagoSerializer(
                caso.ejecutar(request.user.id, estado), many=True).data)

    class GenerarAlertasView(APIView):
        def post(self, request):
            fecha_str = request.data.get("fecha") or request.query_params.get("fecha")
            fecha = date.fromisoformat(fecha_str) if fecha_str else None
            caso = GenerarAlertasDeVencimientoUseCase(
                servicio_fijo_repositorio=DjangoServicioFijoRepository(),
                alerta_repositorio=DjangoAlertaPagoRepository(),
            )
            creadas = caso.ejecutar(fecha)
            return Response(AlertaPagoSerializer(creadas, many=True).data,
                            status=status.HTTP_201_CREATED)

    class AlertaDetailView(APIView):
        def patch(self, request, alerta_id):
            e = CambiarEstadoAlertaSerializer(data=request.data)
            e.is_valid(raise_exception=True)
            repo = DjangoAlertaPagoRepository()
            destino = e.validated_data["estado"]
            caso = (MarcarAlertaComoEnviadaUseCase(repositorio=repo)
                    if destino == "Enviada"
                    else MarcarAlertaComoLeidaUseCase(repositorio=repo))
            try:
                alerta = caso.ejecutar(alerta_id)
            except AlertaNoEncontrada as ex:
                return Response({"detail": str(ex)}, status=status.HTTP_404_NOT_FOUND)
            except TransicionEstadoInvalida as ex:
                return Response({"detail": str(ex)}, status=status.HTTP_409_CONFLICT)
            return Response(AlertaPagoSerializer(alerta).data)

POR QUÉ el PATCH no acepta 'Resuelta': una alerta se resuelve SOLA cuando
se registra el pago (evento GastoRegistrado -> handler). Exponerlo por API
abriría un camino para marcar "resuelta" sin que haya un gasto asociado.
